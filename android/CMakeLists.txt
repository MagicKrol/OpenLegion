cmake_minimum_required(VERSION 3.22)
project(LegionAndroid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(VENDOR ${CMAKE_CURRENT_LIST_DIR}/vendor)
set(ROOT ${CMAKE_CURRENT_LIST_DIR}/..)

set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(${VENDOR}/sdl2 sdl2 EXCLUDE_FROM_ALL)

# sdl12-compat's Linux loader expects a versioned desktop soname. Android
# packages shared objects without a version suffix.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${VENDOR}/sdl12compat/src/SDL12_compat.c)
file(READ ${VENDOR}/sdl12compat/src/SDL12_compat.c SDL12_SOURCE)
string(REPLACE "\"libSDL2-2.0.so.0\"" "\"libSDL2.so\"" SDL12_SOURCE "${SDL12_SOURCE}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/SDL12_compat_android.c "${SDL12_SOURCE}")
add_library(SDL SHARED ${CMAKE_CURRENT_BINARY_DIR}/SDL12_compat_android.c)
target_include_directories(SDL PRIVATE ${VENDOR}/sdl2/include ${VENDOR}/sdl12compat/src)
target_link_libraries(SDL PRIVATE dl)
add_dependencies(SDL SDL2)

set(ANDROID ON CACHE BOOL "" FORCE)
set(NOX11 ON CACHE BOOL "" FORCE)
set(NO_GBM ON CACHE BOOL "" FORCE)
set(EGL_WRAPPER OFF CACHE BOOL "" FORCE)
set(USE_ANDROID_LOG ON CACHE BOOL "" FORCE)
add_subdirectory(${VENDOR}/gl4es gl4es EXCLUDE_FROM_ALL)
set_target_properties(GL PROPERTIES SUFFIX ".so" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}")
target_link_libraries(GL dl m)

set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
add_subdirectory(${VENDOR}/freetype freetype EXCLUDE_FROM_ALL)
set(PNG_SHARED OFF CACHE BOOL "" FORCE)
set(PNG_STATIC ON CACHE BOOL "" FORCE)
set(PNG_TESTS OFF CACHE BOOL "" FORCE)
set(ZLIB_LIBRARY z CACHE STRING "" FORCE)
set(ZLIB_INCLUDE_DIR ${CMAKE_SYSROOT}/usr/include CACHE PATH "" FORCE)
add_subdirectory(${VENDOR}/libpng libpng EXCLUDE_FROM_ALL)
include(ExternalProject)
ExternalProject_Add(jpeg_build SOURCE_DIR ${VENDOR}/jpeg BINARY_DIR ${CMAKE_CURRENT_BINARY_DIR}/jpeg
  CMAKE_ARGS -DCMAKE_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE} -DCMAKE_MAKE_PROGRAM=${CMAKE_MAKE_PROGRAM} -DANDROID_ABI=${ANDROID_ABI}
             -DANDROID_PLATFORM=${ANDROID_PLATFORM} -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
             -DENABLE_SHARED=OFF -DENABLE_STATIC=ON -DWITH_SIMD=OFF
  INSTALL_COMMAND "" BUILD_BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/jpeg/libjpeg.a)
add_library(jpeg-static STATIC IMPORTED GLOBAL)
set_target_properties(jpeg-static PROPERTIES IMPORTED_LOCATION ${CMAKE_CURRENT_BINARY_DIR}/jpeg/libjpeg.a)
add_dependencies(jpeg-static jpeg_build)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/include/libpng12)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/include/libpng12/png.h "#include <png.h>\n")

file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/include/SDL)
foreach(lib sdlttf sdlimage sdlmixer)
  if(lib STREQUAL sdlttf)
    set(header SDL_ttf.h)
  elseif(lib STREQUAL sdlimage)
    set(header SDL_image.h)
  else()
    set(header SDL_mixer.h)
  endif()
  configure_file(${VENDOR}/${lib}/${header} ${CMAKE_CURRENT_BINARY_DIR}/include/SDL/${header} COPYONLY)
endforeach()
set(SDL12_INCLUDES ${VENDOR}/sdl12compat/include ${VENDOR}/sdl12compat/include/SDL ${CMAKE_CURRENT_BINARY_DIR}/include)
add_library(SDL_ttf SHARED ${VENDOR}/sdlttf/SDL_ttf.c)
target_include_directories(SDL_ttf PRIVATE ${SDL12_INCLUDES})
target_link_libraries(SDL_ttf PRIVATE SDL freetype)
file(GLOB IMAGE_SOURCES ${VENDOR}/sdlimage/IMG*.c)
list(FILTER IMAGE_SOURCES EXCLUDE REGEX "IMG_xxx\\.c$")
add_library(SDL_image SHARED ${IMAGE_SOURCES})
target_include_directories(SDL_image PRIVATE ${SDL12_INCLUDES} ${VENDOR}/libpng ${CMAKE_CURRENT_BINARY_DIR}/libpng ${VENDOR}/jpeg ${CMAKE_CURRENT_BINARY_DIR}/jpeg)
target_compile_options(SDL_image PRIVATE -Wno-incompatible-function-pointer-types)
target_compile_definitions(SDL_image PRIVATE LOAD_JPG LOAD_PNG LOAD_BMP LOAD_GIF LOAD_LBM LOAD_PCX LOAD_PNM LOAD_TGA LOAD_XCF LOAD_XPM LOAD_XV)
target_link_libraries(SDL_image PRIVATE SDL png_static jpeg-static z)
set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(ENABLE_STATIC ON CACHE BOOL "" FORCE)
set(ENABLE_SDL OFF CACHE BOOL "" FORCE)
set(ENABLE_ALSA OFF CACHE BOOL "" FORCE)
set(ENABLE_PULSE OFF CACHE BOOL "" FORCE)
add_subdirectory(${VENDOR}/mikmod/libmikmod mikmod EXCLUDE_FROM_ALL)
file(GLOB MIXER_SOURCES ${VENDOR}/sdlmixer/*.c)
list(FILTER MIXER_SOURCES EXCLUDE REGEX "/play(mus|wave)\\.c$")
add_library(SDL_mixer SHARED ${MIXER_SOURCES})
target_include_directories(SDL_mixer PRIVATE ${SDL12_INCLUDES} ${VENDOR}/mikmod/libmikmod/include)
target_compile_options(SDL_mixer PRIVATE -Wno-incompatible-function-pointer-types)
target_compile_definitions(SDL_mixer PRIVATE WAV_MUSIC MOD_MUSIC)
target_link_libraries(SDL_mixer PRIVATE SDL mikmod-static)

file(GLOB GAME_SOURCES CONFIGURE_DEPENDS ${ROOT}/Engine2d/*.cpp ${ROOT}/Engine2d/*.cc ${ROOT}/game/*.cpp)
add_library(main SHARED ${GAME_SOURCES} native/glu_mipmap.c native/android_bridge.cpp)
target_include_directories(main PRIVATE ${ROOT} ${SDL12_INCLUDES} ${VENDOR}/gl4es/include ${VENDOR}/libpng ${CMAKE_CURRENT_BINARY_DIR}/libpng ${VENDOR}/jpeg ${CMAKE_CURRENT_BINARY_DIR}/jpeg)
target_compile_options(main PRIVATE -Wno-deprecated-declarations -Wno-write-strings -Wno-parentheses-equality)
target_link_libraries(main PRIVATE SDL SDL_ttf SDL_image SDL_mixer GL png_static jpeg-static log android z)
